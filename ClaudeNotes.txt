  - Matrix factorization: More like a "curve fitting" problem where you need precise numerical optimization rather than representation learning

  def create_rotated_quadratic_problem(dim=896, condition_number=1000.0, rotation_angle=45):
      """Create a quadratic optimization problem with rotated axes."""

      # Step 1: Create diagonal Hessian (axis-aligned paraboloid)
      eigenvals = torch.logspace(0, np.log10(condition_number), dim)
      H_diag = torch.diag(eigenvals)

      # Step 2: Create rotation matrix (45° in each 2D plane)
      theta = rotation_angle * np.pi / 180
      R = torch.eye(dim)

      # Rotate in each consecutive 2D plane
      for i in range(0, dim-1, 2):
          # 2D rotation matrix
          c, s = np.cos(theta), np.sin(theta)
          R[i:i+2, i:i+2] = torch.tensor([[c, -s], [s, c]])

      # Step 3: Rotated Hessian
      H = R @ H_diag @ R.T

      # Step 4: Define loss function
      def loss_fn(x, target=None):
          if target is None:
              target = torch.zeros_like(x)
          return 0.5 * (x - target).T @ H @ (x - target)

      return H, loss_fn
